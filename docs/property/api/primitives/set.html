properties::set

Synopsis

namespace properties
{
  inline constexpr /* detail */ set{};
}
template<typename M, typename K, typename ...Args>
[[nodiscard]] constexpr
void
operator()(M &&m, K &&k, Args &&...args) const;

Overview

Modifies the property associated to the key k by the property map m using the argument pack args.

Customization point object

A call to properties::set(m, k, args...) is equivalent to:

  1. m.set(k, args...) if it exists,
  2. otherwise, set(m, k, args...), where set is found through argument-dependent lookup (ADL).

Template Parameters

NameDescription
MThe property map type
KThe key type
ArgsThe value type pack

Parameters

NameDescription
mThe property map
kThe key whose associated property is to modify
argsThe value pack

Code Example

#include <map>
#include <memory>
#include <print>
#include <string>
#include <heim/property.hpp>

int main()
{
  std::map<int, std::string>   um{{0, "A"}, {1, "C"}};
  std::unique_ptr<std::string> up{std::make_unique<std::string>("unique_ptr")};

  auto um_map{heim::make_property_map(heim::associative_container_arg, um)};
  auto up_map{heim::make_property_map(heim::pointer_arg)};

  std::print("{} ", um.at(1));
  std::print("{} ", *up);

  heim::properties::set(um_map, 1 , "B");
  heim::properties::set(up_map, up, "unique_ptr");

  std::print("{} ", um.at(1));
  std::print("{} ", *up);
}

Output

C unique_ptr B unique_ptr
