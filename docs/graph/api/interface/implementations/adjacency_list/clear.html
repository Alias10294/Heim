adjacency_list::clear

Synopsis

constexpr
void
clear()
noexcept;

Overview

Erases all vertices and edges from *this.

Complexity

O(V + E), where V and E respectively are the number of vertices and edges in *this.

Code Example

#include <print>
#include <heim/graph.hpp>

template<typename G>
void print_edges(G const &g)
{
  for (auto e : g.edges())
  {
    auto [u, v] = g.endpoints(e);
    std::print("({}, {}) ", u, v);
  }
  std::println();
}

int main()
{
  heim::adjacency_list g{{0, 1}, {0, 2}, {1, 0}, {1, 2}};
  print_edges(g);

  g.clear();
  print_edges(g);
}

Possible Output

(1, 2) (1, 0) (0, 2) (0, 1)
