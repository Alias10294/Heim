graphs::dijkstra_search_result::dijkstra_search_result

Synopsis

constexpr
dijkstra_search_result(
     std::in_place_type_t<G>,
     std::in_place_type_t<WMap>,
     PMap  pm  = {},
     DMap  dm  = {},
     Inf &&inf = std::numeric_limits<Inf>::max(),
     Cmp &&cmp = {});

Overview

Constructs a new dijkstra_search_result using the given property maps.

When property maps are left by default, this object allocates a container for the related properties and automatically generates a property map for it.

Parameters

Name Description
pm The property map used to record reached vertices’ predecessor
dm The property map used to record reached vertices’ distance to their nearest root vertex
inf The value representing the distance to unreached vertices
cmp The object comparing vertices' distances

Complexity

O(1)

Code Example

#include <functional>
#include <limits>
#include <print>
#include <unordered_map>
#include <utility>
#include <heim/graph.hpp>

using graph_type  = heim::adjacency_list<>;
using vertex_type = heim::graphs::vertex_t<graph_type>;
using edge_type   = heim::graphs::edge_t  <graph_type>;

int main()
{
  graph_type                                   g{};
  std::unordered_map<edge_type  , unsigned   > w{};
  std::unordered_map<vertex_type, vertex_type> p{};
  std::unordered_map<vertex_type, unsigned   > d{};

  auto wm{heim::make_property_map(w)};
  auto pm{heim::make_property_map(p)};
  auto dm{heim::make_property_map(d)};

  auto const v0{heim::graphs::place_vertex(g)};
  auto const v1{heim::graphs::place_vertex(g)};
  auto const v2{heim::graphs::place_vertex(g)};
  auto const v3{heim::graphs::place_vertex(g)};

  auto const [e0, placed0]{heim::graphs::place_edge(g, v0, v1)}; w[e0] = 1;
  auto const [e1, placed1]{heim::graphs::place_edge(g, v0, v2)}; w[e1] = 10;
  auto const [e2, placed2]{heim::graphs::place_edge(g, v1, v2)}; w[e2] = 2;
  auto const [e3, placed3]{heim::graphs::place_edge(g, v2, v3)}; w[e3] = 1;

  heim::graphs::dijkstra_result simple {
      std::in_place_type<graph_type>,
      std::in_place_type<decltype(wm)>};
  heim::graphs::dijkstra_result complex{
      std::in_place_type<graph_type>,
      std::in_place_type<decltype(wm)>,
      pm,
      dm,
      std::numeric_limits<unsigned>::max(),
      std::less<unsigned>{}};

  heim::graphs::dijkstra(g, simple , 0, wm);
  heim::graphs::dijkstra(g, complex, 0, wm);
}
